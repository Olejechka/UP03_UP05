# UP03_UP05
```
SELECT 
    з.НомерЗаказа,
    SUM(сз.Кол_во) AS Всего_товаров,
    SUM(сз.Сумма) AS Стоимость_товаров,
    COALESCE(м.Стоимость_материалов, 0) AS Стоимость_материалов,
    SUM(сз.Сумма) + м.Стоимость_материалов AS Полная_стоимость
FROM 
    Заказ з
    JOIN СоставЗаказа сз ON з.НомерЗаказа = сз.НомерЗаказа
    JOIN Продукт п ON сз.Продукт = п.UniqueID
    LEFT JOIN (
        SELECT 
            сз2.НомерЗаказа,
            SUM(сз2.Кол_во * сп.Кол_во * м2.Цена) AS Стоимость_материалов
        FROM 
            СоставЗаказа сз2
            JOIN Продукт п2 ON сз2.Продукт = п2.UniqueID
            JOIN Спецификация сп ON п2.UniqueID = сп.Продукт
            JOIN Материал м2 ON сп.Материал = м2.UniqueID
        GROUP BY 
            сз2.НомерЗаказа
    ) м ON з.НомерЗаказа = м.НомерЗаказа
WHERE 
    з.НомерЗаказа = 1
GROUP BY 
    з.НомерЗаказа, м.Стоимость_материалов;
```
